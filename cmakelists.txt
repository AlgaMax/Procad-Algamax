cmake_minimum_required(VERSION 3.20)
project(ProCAD-GTK VERSION 1.0.0)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Find dependencies
find_package(PkgConfig REQUIRED)
pkg_check_modules(GTK4 REQUIRED gtk4)
pkg_check_modules(CAIRO REQUIRED cairo)
pkg_check_modules(EPOXY REQUIRED epoxy)

# Enable multi-threading
find_package(Threads REQUIRED)

# Source files
file(GLOB_RECURSE SRC_FILES "src/*.cpp")

add_executable(procad_gtk ${SRC_FILES})

target_include_directories(procad_gtk PRIVATE 
    ${CMAKE_SOURCE_DIR}/src
    ${GTK4_INCLUDE_DIRS}
    ${CAIRO_INCLUDE_DIRS}
)

target_link_libraries(procad_gtk
    ${GTK4_LIBRARIES}
    ${CAIRO_LIBRARIES}
    ${EPOXY_LIBRARIES}
    Threads::Threads
)

# Install icon
install(FILES assets/icon.png DESTINATION ${CMAKE_INSTALL_PREFIX}/share/icons)
